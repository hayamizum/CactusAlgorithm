# cactus_old.py の厳密な疑似コード
#
# 用語・記法の厳密な定義：
#   - 作業用グラフ G = (V, E, w)
#       V: 現在の作業用頂点集合
#       E: 現在の作業用辺集合
#       w: E上の重み関数
#   - 出力用グラフ G* = (V*, E*, w*)
#       V*: E*に現れるすべての頂点の集合
#       E*: 確定した辺の集合
#       w*: E*上の重み関数
#   - slack vertex:
#       α(x) = min_{y, z ∈ V, y ≠ z ≠ x} (w(x, y) + w(x, z) - w(y, z)) / 2 > 0 となる x ∈ V
#   - 頂点vの次数:
#       現在の作業用グラフGにおいてvに接続する辺の本数

入力: n次元の距離行列 D （有限集合 X = {0, 1, ..., n-1} 上の距離空間）

1. 初期化
    # 作業用グラフの初期化
    V ← X
    E ← {(i, j) | i ≠ j, i, j ∈ V}
    w(i, j) ← D[i, j]
    # 出力用グラフの初期化
    E* ← ∅
    w* ← ∅
    # V* は E* に現れるすべての頂点の集合として定義する

2. メインループ
    # E または V に変更が生じる限り繰り返す
    繰り返し:

    a. slack vertex の探索と compactification
        # 各 x ∈ V について slack vertex を判定
        各 x ∈ V について：
            α(x) ← min_{y, z ∈ V, y ≠ z ≠ x} (w(x, y) + w(x, z) - w(y, z)) / 2
            (y*, z*) ← α(x) を与える (y, z) のうち、(y, z) のインデックスが最も小さいもの（Vのリスト順で最初のもの）
        S ← { x ∈ V | α(x) > 0 }
        S が空でない場合、S の各 x について：
            新しい頂点 a を導入し、V ← V ∪ {a}
            w(a, x) ← α(x)
            w(a, y*) ← max(w(y*, x) - w(a, x), 0)
            w(a, z*) ← max(w(z*, x) - w(a, x), 0)
            w(a, b) ← max_{t ∈ {x, y*, z*}} (w(b, t) - w(a, t)) for b ∈ V \ {x, y*, z*, a}
            w(x, a) = w(a, x), w(y*, a) = w(a, y*), w(z*, a) = w(a, z*), w(b, a) = w(a, b)
            E ← E ∪ {(a, x), (a, y*), (a, z*)}
        # compactificationを行った場合、再びa.からやり直す

    b. 冗長辺の一括削除
        # 三角不等式が等号で成り立つ冗長な辺を削除
        すべての辺 (i, j) ∈ E について：
            ∃ z ∈ V \ {i, j} で w(i, z) + w(z, j) = w(i, j) かつ w(i, z) ≠ 0 かつ w(z, j) ≠ 0 なら
                E ← E \ {(i, j)}

    c. 確定辺の抽出
        # 次数1の頂点の処理
        G = (V, E, w) において、次数が1の頂点 v ∈ V について：
            vに接続する唯一の辺 (v, u) について：
                E* ← E* ∪ {(v, u)}
                w*(v, u) ← w(v, u)
                E ← E \ {(v, u)}
        # 次数2の頂点の処理
        G = (V, E, w) において、次数が2の頂点 v ∈ V について：
            vに接続する2つの辺 (v, u), (v, w) について：
                E* ← E* ∪ {(v, u), (v, w)}
                w*(v, u) ← w(v, u)
                w*(v, w) ← w(v, w)
                E ← E \ {(v, u), (v, w)}
        # Eにおいてvに接続する辺が存在しない頂点vをVから除外する
        V ← V \ {v | vに接続する辺がEに存在しない}

    d. いずれの操作も適用できない場合、ループを終了する

3. 出力
    # V* = ⋃_{(i, j) ∈ E*} {i, j}
    G* = (V*, E*, w*) を出力する
